name: Build Android APK
on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build-apk:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source Code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Setup Java JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: 17

      - name: Install Capacitor & Android Platform
        run: |
          mkdir -p www
          cp index.html www/index.html || true
          npm init -y
          npm install @capacitor/core @capacitor/cli @capacitor/android
          npx cap add android || true
          npx cap sync android
      - name: Set Custom App Icon (launcher.png)
        run: |
          if [ -f "launcher.png" ]; then
            # Android के सभी mipmap फोल्डर्स में कस्टम लोगो कॉपी करना
            for dir in android/app/src/main/res/mipmap-*; do
              cp launcher.png "$dir/ic_launcher.png"
              cp launcher.png "$dir/ic_launcher_round.png"
              cp launcher.png "$dir/ic_launcher_foreground.png"
            done
            # डिफ़ॉल्ट XML आइकॉन हटाना ताकि आपका PNG लोगो सीधा दिखे
            rm -f android/app/src/main/res/mipmap-anydpi-v26/ic_launcher.xml
            rm -f android/app/src/main/res/mipmap-anydpi-v26/ic_launcher_round.xml
          fi

      - name: Build Debug APK with Gradle
        run: |
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Upload Ready-to-Install APK
        uses: actions/upload-artifact@v4
        with:
          name: Matiks-Offline-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
